const s7cli = async function(input){let base='/Users/yubinai/Documents/jev/.runtime/real-queue';let nonce=input.trial_id+'-'+Date.now();await s7fs.mkdir(base,{recursive:true});await s7fs.writeFile(base+'/input.tmp',JSON.stringify({nonce,input}));await s7fs.rename(base+'/input.tmp',base+'/input.json');let deadline=Date.now()+24000;while(Date.now()<deadline){try{let o=JSON.parse(await s7fs.readFile(base+'/output.json','utf8'));if(o.nonce===nonce){let d=o.decision;d.path='/Users/yubinai/Documents/jev/artifacts/real/decisions/'+(d.artifact||'missing-artifact');return d;}}catch{}await new Promise(resolve=>setTimeout(resolve,50));}throw new Error('QUEUE_RESULT_TIMEOUT');};

const s7reset = async function(task){if(task.startsWith('D')){await resetDyS7();return;}let tab=task==='W3'?alternativeDateStage7:wikiStage7;let url=task==='W3'?'https://www.calculator.net/date-calculator.html':'https://www.wikipedia.org/';let state=await tab.getAXState({emit:false,disableDiffing:true});if(state.includes('URL: "'+url+'"'))await tab.reload();else await tab.goto(url);state=await tab.getAXState({emit:false,disableDiffing:true});if(task==='W2'){if(!state.split('\n').some(x=>/searchLanguage/.test(x)&&/Value: 中文/.test(x))){await tab.playwright.locator('#searchLanguage').selectOption({label:'中文'});state=await tab.getAXState({emit:false,disableDiffing:true});}let obs=s7ObserveModule.observePublic(state,task);if(state.split('\n').some(x=>/ID: searchInput/.test(x)&&/Value:/.test(x))){await tab.setValue(obs.bindings.fill_query,'');await tab.getAXState({emit:false,disableDiffing:true});}}};

const s7runReviewed = async function(task,repeat){let surface=task.startsWith('D')?douyinStage7:task==='W3'?alternativeDateStage7:wikiStage7;let ax='',files=[],events=[];let trial_id=task+'-jev-final-r'+repeat;let started_at=Date.now();let result=await s7Session.runPublicSession({task_id:task,variant:'optimized',trial_id,observe:async()=>{ax=await surface.getAXState({emit:false,disableDiffing:true});return s7ObserveModule.observePublic(ax,task);},decide:async(input)=>{let d=await s7cli(input);files.push(d.path.split('/').pop());if(!d.result){let e=new Error('DECISION_FAILED');e.accounting=d.error?.accounting;throw e;}return d.result;},execute:async(action)=>{await s7AdapterFinal.executePublicControl(surface,action.candidate_id,{[action.candidate_id]:action.binding},ax);},onEvent:async(event)=>{events.push(event);}});let finished_at=Date.now();let row={trial_id,task_id:task,mode:'jev-final',repeat,started_at,finished_at,elapsed_ms:finished_at-started_at,actions:result.actions.length,requests:files.length,status:result.status==='success'?'SUCCEEDED':'FAILED',reason:result.failure_code,decision_files:files,session:result,events};await s7fs.writeFile('/Users/yubinai/Documents/jev/artifacts/real/trials/'+trial_id+'.json',JSON.stringify(row,null,2));nodeRepl.write({trial_id,status:row.status,reason:row.reason,elapsed_ms:row.elapsed_ms,actions:row.actions,requests:row.requests});return row;};
