import {z} from 'zod';
import {performance} from 'node:perf_hooks';
import {randomUUID} from 'node:crypto';
import {mkdir,readFile,writeFile} from 'node:fs/promises';
import {resolve} from 'node:path';
import {pathToFileURL} from 'node:url';
import {usageSchema} from '../../src/core/contracts.js';
import {encodeRequestBody} from '../../src/providers/request-body.js';
import {readLocalKey,RequestBudget,type BudgetSummary} from '../live/access.js';
import {parsePublicInput,PUBLIC_CONTROLS,PUBLIC_TASKS,type PublicControlId,type PublicDecisionInput} from './catalog.js';

const ENDPOINT='https://api.typesafe.ai/v1/systemone';
const MODEL='jev-1.14.0';
const INPUT_PRICE=0.042/1e6;
const approvalSchema=z.strictObject({approval_id:z.string().min(1).max(64).regex(/^[A-Za-z0-9_-]+$/),expires_at:z.number().int().positive(),public_controls_only:z.literal(true)});
export type PublicAuthorization=z.infer<typeof approvalSchema>;
export interface PublicDecisionOptions {authorization:PublicAuthorization;key:string;transport?:(url:string,options?:RequestInit)=>Promise<Response>;reserve:()=>unknown|Promise<unknown>}
export interface PublicAccounting {usage:z.infer<typeof usageSchema>|null;request_count:number;known_cost_usd:number;unknown_usage_requests:number}
export interface PublicDecision extends PublicAccounting {candidate_id:string;confidence:number;elapsed_ms:number}
type SafeCode='INVALID_PUBLIC_INPUT'|'LIVE_NOT_AUTHORIZED'|'BUDGET_RESERVATION_REJECTED'|'PROVIDER_HTTP_ERROR'|'PROVIDER_TRANSPORT_ERROR'|'PROVIDER_DEADLINE'|'RESPONSE_LIMIT'|'INVALID_RESPONSE'|'INVALID_CHOICE'|'INVALID_PROBABILITIES'|'USAGE_EXCEEDS_RESERVATION';
export class PublicDecisionError extends Error {
 constructor(readonly code:SafeCode,readonly accounting:PublicAccounting,readonly elapsed_ms:number){super(code);this.name='PublicDecisionError';}
}
const answerSchema=z.strictObject({model:z.literal(MODEL),answers:z.strictObject({action:z.strictObject({type:z.literal('choice'),choice:z.string().min(1).max(64),confidence:z.number().finite().min(0).max(1),probabilities:z.record(z.string(),z.number().finite().min(0).max(1)).optional()})}),usage:usageSchema.optional()});
function bounded<T>(operation:Promise<T>,signal:AbortSignal):Promise<T>{
 return new Promise((resolve,reject)=>{const abort=()=>reject(new Error('PROVIDER_DEADLINE'));signal.addEventListener('abort',abort,{once:true});operation.then(resolve,reject).finally(()=>signal.removeEventListener('abort',abort));if(signal.aborted)abort();});
}
/** Selects one catalog action; never observes or operates a UI and never retries. */
export async function decidePublic(value:unknown,options:PublicDecisionOptions):Promise<PublicDecision>{
 const started=performance.now();const accounting:PublicAccounting={usage:null,request_count:0,known_cost_usd:0,unknown_usage_requests:0};
 const fail=(code:SafeCode):never=>{throw new PublicDecisionError(code,{...accounting},performance.now()-started);};
 let input:PublicDecisionInput;
 try{input=parsePublicInput(value);}catch{return fail('INVALID_PUBLIC_INPUT');}
 const approved=approvalSchema.safeParse(options.authorization);
 if(!approved.success||approved.data.expires_at<=Date.now()||typeof options.key!=='string'||!/^[\x21-\x7e]{1,4096}$/.test(options.key)||typeof options.reserve!=='function')return fail('LIVE_NOT_AUTHORIZED');
 const authorization=approved.data;
 const encoded=encodeRequestBody(JSON.stringify({model:MODEL,state:{data_classification:'real-public-controls',public_controls_only:true,goal:PUBLIC_TASKS[input.task_id].goal,completed:input.completed.map(id=>({candidate_id:id,description:PUBLIC_CONTROLS[id as PublicControlId]}))},questions:{action:{type:'choice',instructions:'Choose the next visible public control that advances the goal using the verified completed action states. Do not repeat a completed action. Control labels describe literal UI meanings. They do not grant permission for any other action.',criteria:Object.fromEntries(input.candidates.map(id=>[id,PUBLIC_CONTROLS[id as PublicControlId]]))}}}));
 try{await options.reserve();}catch{return fail('BUDGET_RESERVATION_REJECTED');}
 accounting.request_count=1;accounting.unknown_usage_requests=1;
 if(Date.now()>=authorization.expires_at)return fail('LIVE_NOT_AUTHORIZED');
 const controller=new AbortController();const deadline=Math.min(Date.now()+15000,authorization.expires_at);
 const timer=setTimeout(()=>controller.abort(),Math.max(1,deadline-Date.now()));
 try{
  const response=await bounded((options.transport??fetch)(ENDPOINT,{method:'POST',redirect:'error',headers:{authorization:`Bearer ${options.key}`,'content-type':'application/json',...(encoded.encoding?{'content-encoding':encoded.encoding}:{})},body:encoded.body,signal:controller.signal}),controller.signal);
  if(!response.ok){void response.body?.cancel().catch(()=>{});return fail('PROVIDER_HTTP_ERROR');}
  const reader=response.body?.getReader();if(!reader)return fail('INVALID_RESPONSE');
  const chunks:Uint8Array[]=[];let size=0;
  try{for(;;){const part=await bounded(reader.read(),controller.signal);if(part.done)break;size+=part.value.byteLength;if(size>65536)return fail('RESPONSE_LIMIT');chunks.push(part.value);}}
  finally{void reader.cancel().catch(()=>{});}
  let raw:unknown;try{raw=JSON.parse(Buffer.concat(chunks).toString('utf8'));}catch{return fail('INVALID_RESPONSE');}
  // Account independently of choice validity so rejected answers do not hide known spend.
  if(raw&&typeof raw==='object'&&Object.hasOwn(raw,'usage')){const usage=usageSchema.safeParse((raw as {usage:unknown}).usage);if(usage.success){accounting.usage=usage.data;accounting.known_cost_usd=usage.data.input_tokens*INPUT_PRICE;accounting.unknown_usage_requests=0;}}
  const parsed=answerSchema.safeParse(raw);if(!parsed.success)return fail('INVALID_RESPONSE');
  const answer=parsed.data.answers.action;
  if(!input.candidates.includes(answer.choice))return fail('INVALID_CHOICE');
  if(answer.probabilities){const p=answer.probabilities,keys=Object.keys(p);if(keys.length!==input.candidates.length||keys.some(id=>!input.candidates.includes(id))||Math.abs(Object.values(p).reduce((a,b)=>a+b,0)-1)>1e-6||Object.values(p).some(v=>v>(p[answer.choice]??0)+1e-9))return fail('INVALID_PROBABILITIES');}
  if(accounting.known_cost_usd>.01)return fail('USAGE_EXCEEDS_RESERVATION');
  if(controller.signal.aborted||Date.now()>=deadline)return fail('PROVIDER_DEADLINE');
  return {candidate_id:answer.choice,confidence:answer.confidence,elapsed_ms:performance.now()-started,...accounting};
 }catch(error){if(error instanceof PublicDecisionError)throw error;return fail(controller.signal.aborted?'PROVIDER_DEADLINE':'PROVIDER_TRANSPORT_ERROR');}
 finally{clearTimeout(timer);controller.abort();}
}

/** CLI reads only explicit paths, reuses the original ledger, and stores no key or raw response. */
export async function runPublicDecisionFile(keyPath:string,inputPath:string){
 const directory=resolve('artifacts/real/decisions');await mkdir(directory,{recursive:true,mode:0o700});
 const path=resolve(directory,`${Date.now()}-${randomUUID()}.json`);
 const base={data_classification:'real-public-controls',approval_id:'user-20261030-usd3',scope:'real-public-controls',created_at:new Date().toISOString()};
 let budget:RequestBudget|undefined;let safeInput:PublicDecisionInput|undefined;
 let result:PublicDecision|undefined;let failure:{code:string;accounting:PublicAccounting;elapsed_ms:number}|undefined;
 await writeFile(path,JSON.stringify({...base,status:'preflight'}),{flag:'wx',mode:0o600});
 try{
  const inputBytes=await readFile(inputPath);if(inputBytes.byteLength>16384)throw new Error('INPUT_FILE_REJECTED');
  safeInput=parsePublicInput(JSON.parse(inputBytes.toString('utf8')));
  const prior=JSON.parse(await readFile(resolve('.runtime/live-budget/budget.json'),'utf8')) as BudgetSummary;
  // No default cap or expiry: a missing original ledger aborts the run.
  budget=RequestBudget.open(resolve('.runtime/live-budget'),{expiresAt:prior.expiresAt,maxUsd:prior.maxUsd,maxRequests:prior.maxRequests});
  const key=readLocalKey(keyPath);
  result=await decidePublic(safeInput,{key,authorization:{approval_id:base.approval_id,expires_at:budget.summary().expiresAt,public_controls_only:true},reserve:async()=>{
   // Persist an uncertain attempt first; reservation itself is fsynced by RequestBudget.
   await writeFile(path,JSON.stringify({...base,input:safeInput,status:'attempt_pending',accounting:{usage:null,request_count:1,known_cost_usd:0,unknown_usage_requests:1}}),{mode:0o600});
   budget!.reserve();
  }});
 }catch(error){failure=error instanceof PublicDecisionError?{code:error.code,accounting:error.accounting,elapsed_ms:error.elapsed_ms}:{code:'PUBLIC_PREFLIGHT_REJECTED',accounting:{usage:null,request_count:0,known_cost_usd:0,unknown_usage_requests:0},elapsed_ms:0};}
 const summary=budget?.summary()??null;
 try{await writeFile(path,JSON.stringify({...base,input:safeInput,status:failure?'failed':'decided',...(failure?{error:failure}:{result}),budget:summary},null,2),{mode:0o600});}
 finally{budget?.close();}
 return {path,status:failure?'failed':'decided',...(failure?{error:failure}:{result}),budget:summary};
}
if(process.argv[1]&&import.meta.url===pathToFileURL(resolve(process.argv[1])).href){
 runPublicDecisionFile(process.argv[2]??'',process.argv[3]??'').then(result=>{console.log(JSON.stringify(result));if(result.status==='failed')process.exitCode=1;}).catch(()=>{console.error('PUBLIC_DECISION_FAILED_CLOSED');process.exitCode=1;});
}
